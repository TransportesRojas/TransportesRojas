-- 0. EXTENSIONES Y FUNCIONES AUXILIARES
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 1. CATÁLOGOS GEOGRÁFICOS (INEGI AGEEML + SEPOMEX) Y DIRECCIONES

-- 1.1 CATÁLOGO DE ENTIDADES FEDERATIVAS (INEGI)
-- Encabezados origen: CVEGEO, CVE_ENT, NOM_ENT, NOM_ABR, POB_TOTAL, POB_MASCULINA, POB_FEMENINA, TOTAL DE VIVIENDAS HABITADAS
CREATE TABLE cat_estado (
    id          SMALLINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    cvegeo      CHAR(2)  NOT NULL UNIQUE, -- Ej. '01'
    cve_ent     CHAR(2)  NOT NULL UNIQUE, -- Ej. '01'
    nombre      TEXT     NOT NULL,        -- NOM_ENT: 'Aguascalientes'
    abreviatura TEXT     NOT NULL,        -- NOM_ABR: 'Ags.'
    activo      BOOLEAN  NOT NULL DEFAULT TRUE
);

CREATE INDEX idx_cat_estado_nombre ON cat_estado(nombre);

-- 1.2 CATÁLOGO DE MUNICIPIOS (INEGI)
-- Encabezados origen: CVEGEO, CVE_ENT, NOM_ENT, NOM_ABR, CVE_MUN, NOM_MUN, CVE_CAB, NOM_CAB, POB_TOTAL, POB_MASCULINA, POB_FEMENINA, TOTAL DE VIVIENDAS HABITADAS
CREATE TABLE cat_municipio (
    id          INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estado_id   SMALLINT NOT NULL REFERENCES cat_estado(id),
    cvegeo      CHAR(5)  NOT NULL UNIQUE, -- Ej. '01001' (CVE_ENT + CVE_MUN)
    cve_ent     CHAR(2)  NOT NULL,        -- Ej. '01'
    cve_mun     CHAR(3)  NOT NULL,        -- Ej. '001'
    nombre      TEXT     NOT NULL,        -- NOM_MUN: 'Aguascalientes'
    cve_cab     CHAR(4),                  -- CVE_CAB: '0001'
    nom_cab     TEXT,                     -- NOM_CAB: 'Aguascalientes'
    activo      BOOLEAN  NOT NULL DEFAULT TRUE,
    UNIQUE (estado_id, cve_mun),
    UNIQUE (cve_ent, cve_mun)
);

CREATE INDEX idx_cat_municipio_estado ON cat_municipio(estado_id);
CREATE INDEX idx_cat_municipio_nombre ON cat_municipio(nombre);

-- 1.3 CATÁLOGO DE LOCALIDADES (INEGI)
-- Encabezados origen: CVEGEO, Estatus, CVE_ENT, NOM_ENT, NOM_ABR, CVE_MUN, NOM_MUN, CVE_LOC, NOM_LOC, AMBITO, LATITUD, LONGITUD, LAT_DECIMAL, LON_DECIMAL, ALTITUD, CVE_CARTA, POB_TOTAL, POB_MASCULINA, POB_FEMENINA, TOTAL DE VIVIENDAS HABITADAS
CREATE TABLE cat_localidad (
    id           BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estado_id    SMALLINT     NOT NULL REFERENCES cat_estado(id),
    municipio_id INT          NOT NULL REFERENCES cat_municipio(id),
    cvegeo       CHAR(9)      NOT NULL UNIQUE, -- Ej. '010010001' (CVE_ENT + CVE_MUN + CVE_LOC)
    estatus      TEXT,                         -- Columna 'Estatus'
    cve_ent      CHAR(2)      NOT NULL,        -- Ej. '01'
    cve_mun      CHAR(3)      NOT NULL,        -- Ej. '001'
    cve_loc      CHAR(4)      NOT NULL,        -- Ej. '0001'
    nombre       TEXT         NOT NULL,        -- NOM_LOC: 'AGUASCALIENTES'
    ambito       CHAR(1)      CHECK (ambito IN ('U', 'R')), -- 'U' = Urbano, 'R' = Rural
    lat_decimal  NUMERIC(10,7),                -- LAT_DECIMAL: 21.8798220
    lon_decimal  NUMERIC(10,7),                -- LON_DECIMAL: -102.2960460
    altitud      INT,                          -- ALTITUD: 1878
    activo       BOOLEAN      NOT NULL DEFAULT TRUE,
    UNIQUE (municipio_id, cve_loc),
    UNIQUE (cve_ent, cve_mun, cve_loc)
);

CREATE INDEX idx_cat_localidad_municipio ON cat_localidad(municipio_id);
CREATE INDEX idx_cat_localidad_estado    ON cat_localidad(estado_id);
CREATE INDEX idx_cat_localidad_nombre    ON cat_localidad(nombre);

-- 1.4 CATÁLOGO DE CÓDIGOS POSTALES Y ASENTAMIENTOS / COLONIAS (SEPOMEX)
-- Encabezados origen: d_codigo, d_asenta, d_tipo_asenta, D_mnpio, d_estado, d_ciudad, d_CP, c_estado, c_oficina, c_CP, c_tipo_asenta, c_mnpio, id_asenta_cpcons, d_zona, c_cve_ciudad
CREATE TABLE cat_asentamiento (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estado_id        SMALLINT NOT NULL REFERENCES cat_estado(id),
    municipio_id     INT      NOT NULL REFERENCES cat_municipio(id),
    d_codigo         CHAR(5)  NOT NULL, -- Código Postal del asentamiento (Ej. '20000')
    d_asenta         TEXT     NOT NULL, -- Nombre de la colonia/asentamiento (Ej. 'Aguascalientes Centro')
    d_tipo_asenta    TEXT     NOT NULL, -- Tipo de asentamiento (Ej. 'Colonia', 'Fraccionamiento')
    d_ciudad         TEXT,              -- Nombre de la ciudad (Ej. 'Aguascalientes')
    d_cp             CHAR(5),           -- CP de la Administración Postal que reparte (Ej. '20001')
    c_estado         CHAR(2)  NOT NULL, -- Clave Entidad INEGI (Ej. '01')
    c_oficina        CHAR(5),           -- Clave de oficina postal (Ej. '20001')
    c_cp             TEXT,              -- Campo c_CP del catálogo SEPOMEX
    c_tipo_asenta    CHAR(2),           -- Clave tipo de asentamiento (Ej. '09')
    c_mnpio          CHAR(3)  NOT NULL, -- Clave Municipio INEGI (Ej. '001')
    id_asenta_cpcons CHAR(4)  NOT NULL, -- Identificador único del asentamiento a nivel municipal (Ej. '0001')
    d_zona           TEXT,              -- Zona: 'Urbano', 'Rural', 'Semiurbano'
    c_cve_ciudad     CHAR(2),           -- Clave de ciudad (Ej. '01')
    activo           BOOLEAN  NOT NULL DEFAULT TRUE,
    UNIQUE (c_estado, c_mnpio, id_asenta_cpcons, d_codigo)
);

CREATE INDEX idx_cat_asentamiento_cp        ON cat_asentamiento(d_codigo);
CREATE INDEX idx_cat_asentamiento_municipio ON cat_asentamiento(municipio_id);
CREATE INDEX idx_cat_asentamiento_estado    ON cat_asentamiento(estado_id);
CREATE INDEX idx_cat_asentamiento_nombre    ON cat_asentamiento(d_asenta);

    localidad_id    BIGINT   REFERENCES cat_localidad(id),
    asentamiento_id BIGINT   REFERENCES cat_asentamiento(id),

    -- Datos de vialidad y domicilio
    calle           TEXT,
    numero_exterior TEXT,
    numero_interior TEXT,
    entre_calles    TEXT,

    -- Campos textuales (se autocompletan desde los catálogos si se envían los IDs, o sirven de respaldo)

    estado          TEXT,

-- Trigger para mantener sincronizados los nombres y coordenadas por defecto en direccion
CREATE OR REPLACE FUNCTION trg_fn_direccion_sincronizar_catalogo()
RETURNS TRIGGER AS $$
DECLARE
    v_asenta RECORD;
    v_loc    RECORD;
    v_mun    RECORD;
    v_est    RECORD;
BEGIN
    -- 1. Si se especifica asentamiento_id (SEPOMEX), autocompletar colonia, CP, municipio_id y estado_id
    IF NEW.asentamiento_id IS NOT NULL THEN
        SELECT d_asenta, d_codigo, municipio_id, estado_id, d_ciudad
          INTO v_asenta
          FROM cat_asentamiento
         WHERE id = NEW.asentamiento_id;

        IF FOUND THEN
            NEW.colonia       := COALESCE(NEW.colonia, v_asenta.d_asenta);
            NEW.codigo_postal := COALESCE(NEW.codigo_postal, v_asenta.d_codigo);
            NEW.municipio_id  := COALESCE(NEW.municipio_id, v_asenta.municipio_id);
            NEW.estado_id     := COALESCE(NEW.estado_id, v_asenta.estado_id);
            NEW.localidad     := COALESCE(NEW.localidad, v_asenta.d_ciudad);
        END IF;
    END IF;

    -- 2. Si se especifica localidad_id (INEGI), autocompletar localidad, coordenadas por defecto, municipio_id y estado_id
    IF NEW.localidad_id IS NOT NULL THEN
        SELECT nombre, lat_decimal, lon_decimal, municipio_id, estado_id
          INTO v_loc
          FROM cat_localidad
         WHERE id = NEW.localidad_id;

        IF FOUND THEN
            NEW.localidad    := COALESCE(NEW.localidad, v_loc.nombre);
            NEW.latitud      := COALESCE(NEW.latitud, v_loc.lat_decimal);
            NEW.longitud     := COALESCE(NEW.longitud, v_loc.lon_decimal);
            NEW.municipio_id := COALESCE(NEW.municipio_id, v_loc.municipio_id);
            NEW.estado_id    := COALESCE(NEW.estado_id, v_loc.estado_id);
        END IF;
    END IF;

    -- 3. Si se especifica municipio_id, autocompletar nombre de municipio y estado_id
    IF NEW.municipio_id IS NOT NULL THEN
        SELECT nombre, estado_id
          INTO v_mun
          FROM cat_municipio
         WHERE id = NEW.municipio_id;

        IF FOUND THEN
            NEW.municipio := COALESCE(NEW.municipio, v_mun.nombre);
            NEW.estado_id := COALESCE(NEW.estado_id, v_mun.estado_id);
        END IF;
    END IF;

    -- 4. Si se especifica estado_id, autocompletar nombre de estado
    IF NEW.estado_id IS NOT NULL THEN
        SELECT nombre
          INTO v_est
          FROM cat_estado
         WHERE id = NEW.estado_id;

        IF FOUND THEN
            NEW.estado := COALESCE(NEW.estado, v_est.nombre);
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_direccion_sincronizar_catalogo
BEFORE INSERT OR UPDATE ON direccion
FOR EACH ROW EXECUTE FUNCTION trg_fn_direccion_sincronizar_catalogo();

CREATE TRIGGER trg_direccion_updated BEFORE UPDATE ON direccion FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- SUCURSAL (Sin el campo de horario de texto, ese se maneja en tabla aparte)

    nombre_corto      TEXT, 
    codigo            TEXT UNIQUE,

    updated_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

-- COBERTURA DE SUCURSAL A NIVEL NACIONAL (Permite determinar qué sucursal atiende cada estado/municipio/CP)
CREATE TABLE sucursal_cobertura (
    id                 BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    sucursal_id        BIGINT   NOT NULL REFERENCES sucursal(id) ON DELETE CASCADE,
    estado_id          SMALLINT NOT NULL REFERENCES cat_estado(id),
    municipio_id       INT      REFERENCES cat_municipio(id), -- NULL = cubre todo el estado
    codigo_postal      CHAR(5),                               -- NULL = cubre todo el municipio/estado
    aplica_entrega     BOOLEAN  NOT NULL DEFAULT TRUE,
    es_zona_extendida  BOOLEAN  NOT NULL DEFAULT FALSE,
    dias_estimados     SMALLINT NOT NULL DEFAULT 1 CHECK (dias_estimados > 0),
    activo             BOOLEAN  NOT NULL DEFAULT TRUE,
    created_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE NULLS NOT DISTINCT (sucursal_id, estado_id, municipio_id, codigo_postal)
);

CREATE INDEX idx_suc_cobertura_sucursal  ON sucursal_cobertura(sucursal_id);
CREATE INDEX idx_suc_cobertura_estado    ON sucursal_cobertura(estado_id);
CREATE INDEX idx_suc_cobertura_municipio ON sucursal_cobertura(municipio_id);
CREATE INDEX idx_suc_cobertura_cp        ON sucursal_cobertura(codigo_postal);

CREATE TABLE horario_sucursal (
    id              BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    sucursal_id     BIGINT NOT NULL REFERENCES sucursal(id) ON DELETE CASCADE,
    dia_semana      INT NOT NULL CHECK (dia_semana BETWEEN 1 AND 7), -- 1=Lunes, 7=Domingo
    hora_apertura   TIME,
    hora_cierre     TIME,
    es_dia_descanso BOOLEAN NOT NULL DEFAULT FALSE,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    -- Validación: Si no es descanso, debe haber hora de apertura y cierre válida.
    -- Si es descanso, las horas deben ser nulas.
    CONSTRAINT chk_horario_valido CHECK (
        (es_dia_descanso = TRUE AND hora_apertura IS NULL AND hora_cierre IS NULL)
        OR
        (es_dia_descanso = FALSE AND hora_apertura IS NOT NULL AND hora_cierre IS NOT NULL AND hora_apertura < hora_cierre)
    )
);

CREATE INDEX idx_horario_sucursal ON horario_sucursal(sucursal_id);

-- 11. TRIGGERS DE MANTENIMIENTO
CREATE OR REPLACE FUNCTION trg_fn_ruta_actualizar_odometro()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.estado = 'finalizada' AND NEW.km_final_odometro IS NOT NULL AND NEW.unidad_id IS NOT NULL AND (OLD.estado <> 'finalizada' OR OLD.km_final_odometro IS DISTINCT FROM NEW.km_final_odometro) THEN
        UPDATE unidad SET km_odometro_actual = NEW.km_final_odometro WHERE id = NEW.unidad_id AND NEW.km_final_odometro > km_odometro_actual;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_ruta_actualizar_odometro AFTER UPDATE ON ruta FOR EACH ROW EXECUTE FUNCTION trg_fn_ruta_actualizar_odometro();

CREATE OR REPLACE FUNCTION trg_fn_generar_alerta_mantenimiento()
RETURNS TRIGGER AS $$
DECLARE
    tm RECORD;
    v_km_ultimo_mantenimiento NUMERIC(10,2);
    v_km_base NUMERIC(10,2);
BEGIN
    IF NEW.km_odometro_actual IS NOT DISTINCT FROM OLD.km_odometro_actual THEN RETURN NEW; END IF;

    FOR tm IN SELECT * FROM tipo_mantenimiento WHERE activo = TRUE AND (aplica_a_tipo_vehiculo IS NULL OR aplica_a_tipo_vehiculo = NEW.tipo_vehiculo)
    LOOP
        SELECT MAX(km_al_momento) INTO v_km_ultimo_mantenimiento FROM mantenimiento WHERE unidad_id = NEW.id AND tipo_mantenimiento_id = tm.id;
        v_km_base := COALESCE(v_km_ultimo_mantenimiento, COALESCE(NEW.km_odometro_inicial, 0));

        IF (NEW.km_odometro_actual - v_km_base) >= tm.km_intervalo THEN
            IF NOT EXISTS (SELECT 1 FROM alerta_mantenimiento am WHERE am.unidad_id = NEW.id AND am.tipo_mantenimiento_id = tm.id AND am.estado IN ('nueva','vista')) THEN
                INSERT INTO alerta_mantenimiento (unidad_id, tipo_mantenimiento_id, km_al_momento, km_umbral, km_ultimo_mantenimiento, descripcion, estado) 
                VALUES (NEW.id, tm.id, NEW.km_odometro_actual, v_km_base + tm.km_intervalo, v_km_ultimo_mantenimiento, 'La unidad requiere ' || tm.nombre || ' (intervalo: ' || tm.km_intervalo || ' km)', 'nueva');
            END IF;
        END IF;
    END LOOP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_unidad_generar_alerta_mantenimiento AFTER UPDATE ON unidad FOR EACH ROW EXECUTE FUNCTION trg_fn_generar_alerta_mantenimiento();
